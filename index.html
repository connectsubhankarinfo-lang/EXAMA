// ========================================================== // EXAMA – Unified Google Drive uploader (v2: music + list + delete) // One script handles every upload type. The app sends a // "folderType" key and this script picks the right folder. // // NEW in v2 // * 'music' folder -> Run Monitor songs (MP3 / MP4 only) // * doGet ?action=list -> returns all songs in the music folder // * doPost {action:'delete', fileId} -> removes a file (only from the folders below) // * music uploads keep song name, artist and artwork (sent by the app, read from the file tags) // ========================================================== var FOLDERS = { 'dp': '1RebqK80V-oQOOr-pws36wfIU8adOOxmp', // student profile picture 'cover': '1LzPZSNATSpViwEkYwKuFOW_IbGdOW8fD', // student cover photo 'run': '1nxKeGfCi4CroO_TyBPJdgP8MT6CF4DI0', // run monitor image 'material': '1H5gAsnFbeK-ksss9K8mtRslnt3pcQ4Q0', // material hub upload 'script': '1T7SKgAqDELPw9hxcEL29ODoGZCq4gcgB', // P2P exam answer script 'qa': '1IFrw3Xi6OIXNIrpSyTZKp1bQQ5QFWR3P', // P2P exam questions 'annotation': '1kQZgBXqmjw2nvUVdrLDgHKf0NNvrAxvY', // red pen annotation save 'chat_img': '13jOtkrws2U25UcBKYebnYUFLxvpA48mi', // discuss image 'chat_doc': '10smsLb4KOw4Nv6mofuzpMBNmqtdqE1JQ', // discuss documents 'notice': '1HHSLvIMgWmEAZUbo0NwDNbw23SY6B3ov', // notice board image 'music': '1p-u26mIgor6A4obiNUP9Ujd4FsNOcFvA' // run monitor music (mp3 / mp4) + their artwork }; // Folder used if an unknown folderType arrives (keeps old cached app versions working) var FALLBACK_FOLDER = 'chat_img'; // These folder types return a direct-download link, everything else returns an image thumbnail link var DOWNLOAD_TYPES = ['chat_doc', 'material', 'script', 'qa', 'music']; // Only these extensions are accepted in the music folder var MUSIC_EXT = ['mp3', 'mp4']; var LIST_CACHE_KEY = 'music_list_v1'; // ---------------------------------------------------------- helpers function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } function extOf_(name) { var p = String(name || '').split('.'); return p.length > 1 ? p.pop().toLowerCase() : ''; } function thumb_(id) { return 'https://drive.google.com/thumbnail?id=' + id + '&sz=w400'; } function shareView_(file) { try { file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (shareErr) { // folder permission is inherited anyway, so ignore } } // A file may only be deleted by the app if it lives inside one of the Exama folders above function inKnownFolder_(file) { var ids = Object.keys(FOLDERS).map(function (k) { return FOLDERS[k]; }); var parents = file.getParents(); while (parents.hasNext()) { if (ids.indexOf(parents.next().getId()) > -1) return true; } return false; } // Permanent delete when the "Drive API" advanced service is switched on, otherwise move to trash function kill_(file) { try { Drive.Files.remove(file.getId()); } catch (e) { file.setTrashed(true); } } function clearListCache_() { try { CacheService.getScriptCache().remove(LIST_CACHE_KEY); } catch (e) { } } // ---------------------------------------------------------- POST: upload / delete function doPost(e) { try { if (!e) throw new Error("No data received."); // Supports both: JSON body (fetch) and form field "payload=" (material uploader) var raw = null; if (e.parameter && e.parameter.payload) { raw = e.parameter.payload; } else if (e.postData && e.postData.contents) { raw = e.postData.contents; } if (!raw) throw new Error("No data received."); var data = JSON.parse(raw); if (data.action === 'delete') return json_(deleteFile_(data)); return json_(uploadFile_(data)); } catch (err) { return json_({ success: false, error: err.toString() }); } } function uploadFile_(data) { var base64 = data.base64; if (!base64) throw new Error("No file content (base64) received."); var filename = data.fileName || ("Exama_File_" + new Date().getTime()); var mimeType = data.mimeType || 'image/jpeg'; var folderType = data.folderType || FALLBACK_FOLDER; if (base64.indexOf(',') > -1) { base64 = base64.split(',')[1]; } var isMusic = (folderType === 'music'); if (isMusic) { if (MUSIC_EXT.indexOf(extOf_(filename)) < 0) throw new Error("Only MP3 and MP4 files can be added."); mimeType = (extOf_(filename) === 'mp3') ? 'audio/mpeg' : 'video/mp4'; } var folderId = FOLDERS[folderType] || FOLDERS[FALLBACK_FOLDER]; var folder = DriveApp.getFolderById(folderId); var blob = Utilities.newBlob(Utilities.base64Decode(base64), mimeType, filename); var file = folder.createFile(blob); // Make sure the file is viewable by link even if the folder sharing was missed shareView_(file); // ---- music: keep the song name, artist and artwork next to the file ---- if (isMusic) { var title = String(data.title || '').substring(0, 200); var artist = String(data.artist || '').substring(0, 200); var artId = ''; if (data.artBase64) { try { var artBlob = Utilities.newBlob(Utilities.base64Decode(data.artBase64), 'image/jpeg', 'art_' + file.getId() + '.jpg'); var artFile = folder.createFile(artBlob); shareView_(artFile); artId = artFile.getId(); } catch (artErr) { artId = ''; // song is saved anyway, just without artwork } } file.setDescription(JSON.stringify({ t: title, a: artist, art: artId })); clearListCache_(); return { success: true, id: file.getId(), url: "https://drive.google.com/uc?id=" + file.getId() + "&export=download", title: title, artist: artist, art: artId ? thumb_(artId) : '', mime: mimeType, folderType: folderType }; } var finalUrl; if (mimeType === 'application/pdf' || DOWNLOAD_TYPES.indexOf(folderType) > -1) { finalUrl = "https://drive.google.com/uc?id=" + file.getId() + "&export=download"; } else { finalUrl = "https://drive.google.com/thumbnail?id=" + file.getId() + "&sz=w1200"; } return { success: true, url: finalUrl, id: file.getId(), folderType: folderType }; } function deleteFile_(data) { var id = String(data.fileId || '').replace(/[^\w-]/g, ''); if (!id) throw new Error("No file id."); var file; try { file = DriveApp.getFileById(id); } catch (e) { // already gone: treat as success so the app can carry on return { success: true, id: id, gone: true }; } if (!inKnownFolder_(file)) throw new Error("This file is not managed by Exama."); // a song also owns an artwork file: remove that one too try { var meta = JSON.parse(file.getDescription() || '{}'); if (meta && meta.art) { var art = DriveApp.getFileById(meta.art); if (inKnownFolder_(art)) kill_(art); } } catch (e) { } kill_(file); clearListCache_(); return { success: true, id: id }; } // ---------------------------------------------------------- GET: health check + song list function listSongs_() { var cache = CacheService.getScriptCache(); var hit = cache.get(LIST_CACHE_KEY); if (hit) { try { return JSON.parse(hit); } catch (e) { } } var out = []; var it = DriveApp.getFolderById(FOLDERS['music']).getFiles(); while (it.hasNext()) { var f = it.next(); var name = f.getName(); if (MUSIC_EXT.indexOf(extOf_(name)) < 0) continue; // skips artwork images and anything else var meta = {}; try { meta = JSON.parse(f.getDescription() || '{}') || {}; } catch (e) { meta = {}; } out.push({ id: f.getId(), name: name, title: meta.t || '', artist: meta.a || '', art: meta.art ? thumb_(meta.art) : '', mime: f.getMimeType(), size: f.getSize(), created: f.getDateCreated().getTime() }); } out.sort(function (a, b) { return b.created - a.created; }); // newest first try { cache.put(LIST_CACHE_KEY, JSON.stringify(out), 120); } catch (e) { } return out; } // Quick health check: open the web app URL in a browser, you should see {"ok":true} function doGet(e) { try { if (e && e.parameter && e.parameter.action === 'list') { return json_({ success: true, files: listSongs_() }); } } catch (err) { return json_({ success: false, error: err.toString() }); } return json_({ ok: true, service: "Exama Drive uploader", version: 2 }); } function doOptions(e) { return ContentService.createTextOutput("").setMimeType(ContentService.MimeType.TEXT); }
